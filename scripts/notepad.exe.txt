//----- Key pressed events --------------------------------------------------------------------------------------------

WhenKeyPressed(left arrow)
{
    If (domemory.exe == 2)
    {
        cursor location += -1;
        If (cursor location == 0)
        {
            cursor location += 1;
        }
    }
}

WhenKeyPressed(right arrow)
{
    If (domemory.exe == 2)
    {
        If ((EDITMODE < 3) Or (EDITMODE == 6))
        {
            cursor location += 1;
            If (cursor location > (Variable.Length(TextEdit) + 1))
            {
                cursor location += -1;
            }
        }
    }
}

Define INSERT at character(string chr) string(string NEW) oldstring(string OLD) (warp=true)
{
    cccccc = 0;
    instring = "";
    Repeat (chr)
    {
        cccccc += 1;
        instring = Operator.Join(instring, Operator.LetterOf(OLD, cccccc));
    }
    instring = Operator.Join(instring, NEW);
    Repeat ((Variable.Length(OLD) - chr))
    {
        cccccc += 1;
        instring = Operator.Join(instring, Operator.LetterOf(OLD, cccccc));
    }
}
