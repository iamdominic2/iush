//----- Key pressed events --------------------------------------------------------------------------------------------

WhenKeyPressed(left arrow)
{
    If ((EDITMODE < 3) Or (EDITMODE == 6))
    {
        cursor location += -1;
        If (cursor location == 0)
        {
            cursor location += 1;
        }
    }
}

WhenKeyPressed(right arrow)
{
    If ((EDITMODE < 3) Or (EDITMODE == 6))
    {
        cursor location += 1;
        If (cursor location > (Variable.Length(TextEdit) + 1))
        {
            cursor location += -1;
        }
    }
}

WhenKeyPressed(up arrow)
{
    If Sensing.KeyPressed(up arrow)
    {
        If (counta > 1)
        {
            counta += -1;
            TextEdit = recents[counta];
            cursor location = (Variable.Length(recents[counta]) + 1);
        }
    }
    Else
    {
        If (scroll > 0)
        {
            scroll += -1;
        }
        Else
        {
        }
    }
}

WhenKeyPressed(v)
{
    If Sensing.KeyPressed(Data[List.IndexOf(Name, "C:\registry\controlkey")])
    {
        Event.Broadcast("PASTE");
    }
}

WhenKeyPressed(down arrow)
{
    If Sensing.KeyPressed(down arrow)
    {
        If ((counta - 1) < List.Length(recents))
        {
            counta += 1;
            TextEdit = recents[counta];
            cursor location = (Variable.Length(recents[counta]) + 1);
        }
    }
    Else
    {
        If (scroll < List.Length(cm))
        {
            scroll += 1;
        }
        Else
        {
        }
    }
}

WhenKeyPressed(x)
{
    List.Hide(Name);
    List.Hide(Data);
}

WhenKeyPressed(page up)

WhenKeyPressed(page down)

    }
}
